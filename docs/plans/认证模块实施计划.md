```php
<?php
// app/Http/Controllers/Api/CliAuthController.php
namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Tymon\JWTAuth\Facades\JWTAuth;

class CliAuthController extends Controller
{
    /**
     * 使用 session_id 换取当前系统的 JWT token
     * CLI 专用接口，不需要 JWT 中间件
     */
    public function token(Request $request)
    {
        $sessionId = $request->input('session_id');

        if (!$sessionId) {
            return response()->json([
                'error' => 'session_id 不能为空',
            ], 422);
        }

        // 验证 session_id 有效性
        $session = DB::connection('ops_user')
            ->table('shared_login_sessions')
            ->where('session_id', $sessionId)
            ->where('expires_at', '>', date('Y-m-d H:i:s'))
            ->where('is_valid', 1)
            ->first();

        if (!$session) {
            return response()->json([
                'error' => 'session_invalid',
                'message' => 'session 已过期或无效，请重新运行:opscli login',
            ], 401);
        }

        // 根据 email 查找用户
        $user = \App\Models\Sys\UserOrm::where('email', $session->email)->first();

        if (!$user) {
            return response()->json([
                'error' => 'user_not_found',
                'message' => '用户不存在',
            ], 404);
        }

        // 生成 JWT token
        $token = JWTAuth::fromUser($user);
        $ttl = config('jwt.ttl', 120); // 分钟

        return response()->json([
            'jwt' => $token,
            'expires_in' => $ttl * 60, // 转为秒
        ]);
    }
}
```

```php
<?php
// app/Http/Controllers/Api/CliDeviceController.php
namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;

class CliDeviceController extends Controller
{
    private const CODE_TTL = 300;      // 设备码有效期（秒）
    private const POLL_INTERVAL = 3;   // 建议轮询间隔（秒）

    /**
     * CLI 申请设备码
     * POST /v1/cli/device/code
     */
    public function requestCode(Request $request)
    {
        $deviceCode = Str::uuid()->toString();
        $userCode = strtoupper(Str::random(4) . '-' . Str::random(4));
        $expiresAt = now()->addSeconds(self::CODE_TTL);

        DB::table('cli_device_codes')->insert([
            'device_code' => $deviceCode,
            'user_code'   => $userCode,
            'expires_at'  => $expiresAt,
            'status'      => 'pending',
            'created_at'  => now(),
            'updated_at'  => now(),
        ]);

        return response()->json([
            'device_code'      => $deviceCode,
            'user_code'        => $userCode,
            'verification_url' => config('app.url') . '/cli-auth',
            'expires_in'       => self::CODE_TTL,
            'interval'         => self::POLL_INTERVAL,
        ]);
    }

    /**
     * CLI 轮询授权结果
     * GET /v1/cli/device/poll?device_code=xxx
     */
    public function poll(Request $request)
    {
        $deviceCode = $request->input('device_code');

        $record = DB::table('cli_device_codes')
            ->where('device_code', $deviceCode)
            ->first();

        if (!$record) {
            return response()->json(['status' => 'expired'], 200);
        }

        // 检查是否已过期
        if (now()->isAfter($record->expires_at)) {
            DB::table('cli_device_codes')
                ->where('device_code', $deviceCode)
                ->update(['status' => 'expired', 'updated_at' => now()]);
            return response()->json(['status' => 'expired'], 200);
        }

        if ($record->status === 'authorized') {
            // 查询 session 过期时间
            $session = DB::connection('ops_user')
                ->table('shared_login_sessions')
                ->where('session_id', $record->session_id)
                ->first();

            return response()->json([
                'status'     => 'authorized',
                'session_id' => $record->session_id,
                'email'      => $record->email,
                'expires_at' => $session->expires_at ?? '',
            ]);
        }

        return response()->json(['status' => $record->status]);
    }

    /**
     * 用户在浏览器确认授权（需登录态，由 JwtAuthMiddleware 保护）
     * POST /v1/cli/device/confirm
     */
    public function confirm(Request $request)
    {
        $userCode = strtoupper($request->input('user_code', ''));

        $record = DB::table('cli_device_codes')
            ->where('user_code', $userCode)
            ->where('status', 'pending')
            ->first();

        if (!$record || now()->isAfter($record->expires_at)) {
            return response()->json([
                'success' => false,
                'message' => '验证码无效或已过期',
            ], 422);
        }

        // 获取当前登录用户的 session_id（从 polarisUserToken cookie）
        $sessionId = $request->cookie('polarisUserToken');
        $user = $request->user(); // JwtAuthMiddleware 注入

        DB::table('cli_device_codes')
            ->where('user_code', $userCode)
            ->update([
                'session_id' => $sessionId,
                'email'      => $user->email ?? '',
                'status'     => 'authorized',
                'updated_at' => now(),
            ]);

        return response()->json(['success' => true]);
    }

    /**
     * 拒绝授权（可选）
     * POST /v1/cli/device/deny
     */
    public function deny(Request $request)
    {
        $userCode = strtoupper($request->input('user_code', ''));

        DB::table('cli_device_codes')
            ->where('user_code', $userCode)
            ->where('status', 'pending')
            ->update(['status' => 'denied', 'updated_at' => now()]);

        return response()->json(['success' => true]);
    }

    /**
     * 获取多实例系统列表（一期预留，返回空列表）
     * GET /v1/cli/systems
     */
    public function systems(Request $request)
    {
        // 一期仅 ops+bi，数据分析系统二期在此动态返回
        return response()->json(['systems' => []]);
    }
}
```

```php
namespace App\Http\Controllers;

     * 使用 session_id 换取当前系统的 JWT token

    public function token(Request $request)
    {
        $sessionId = $request->input('session_id');

            ], 401);

        if (!$user) {
            return response()->json([
                'error'   => 'user_not_found',
                'message' => '该邮箱在本系统中不存在',
            ], 404);

        $ttl = config('jwt.ttl', 120);

        return response()->json([
            'jwt'        => $token,
            'expires_in' => $ttl * 60,
        ]);
    }
}
```

**Step 3: 创建 exceptions.py**

```python
class AuthError(Exception):
    """基础认证异常"""

class NotAuthenticatedError(AuthError):
    """未登录或 session 已失效"""

class SessionExpiredError(AuthError):
    """session_id 已过期"""

class TokenFetchError(AuthError):
    """从业务系统获取 JWT 失败"""

class SystemNotFoundError(AuthError):
    """系统别名未注册"""

class DeviceFlowError(AuthError):
```

```python
# opscli/storage/crypto.py
import os, stat
from pathlib import Path
from cryptography.hazmat.primitives.ciphers.aead import AESGCM

class Crypto:
    def __init__(self, key_path: Path):
        self._key_path = Path(key_path)
        self._key = self._load_or_create_key()

    def _load_or_create_key(self) -> bytes:
        if self._key_path.exists():
            return self._key_path.read_bytes()
        key = AESGCM.generate_key(bit_length=256)
        self._key_path.write_bytes(key)
        self._key_path.chmod(stat.S_IRUSR | stat.S_IWUSR)
        return key

    def encrypt(self, plaintext: str) -> bytes:
        nonce = os.urandom(12)
        return nonce + AESGCM(self._key).encrypt(nonce, plaintext.encode(), None)

    def decrypt(self, ciphertext: bytes) -> str:
        return AESGCM(self._key).decrypt(ciphertext[:12], ciphertext[12:], None).decode()
```

```python
# opscli/storage/credential_store.py
import json, time, stat
from datetime import datetime, timezone, timedelta
from pathlib import Path
from opscli.storage.crypto import Crypto

class CredentialStore:
    def __init__(self, base_dir: Path | None = None):
        self._dir = Path(base_dir or (Path.home() / ".config" / "opscli"))
        self._dir.mkdir(parents=True, exist_ok=True)
        self._path = self._dir / "credentials.bin"
        self._crypto = Crypto(self._dir / ".key")

    def load(self) -> dict | None:
        if not self._path.exists():
            return None
        return json.loads(self._crypto.decrypt(self._path.read_bytes()))

    def _save(self, data: dict):
        self._path.write_bytes(self._crypto.encrypt(json.dumps(data)))
        self._path.chmod(stat.S_IRUSR | stat.S_IWUSR)

    def save_session(self, session_id: str, email: str, expires_at: str):
        data = self.load() or {}
        data.update({"session_id": session_id, "email": email,
                     "session_expires_at": expires_at,
                     "tokens": data.get("tokens", {})})
        self._save(data)

    def save_token(self, system_key: str, jwt: str, expires_in: int):
        data = self.load() or {}
        data.setdefault("tokens", {})
        expires_at = (datetime.now(timezone.utc) + timedelta(seconds=expires_in)).isoformat()
        data["tokens"][system_key] = {"jwt": jwt, "expires_at": expires_at,
                                       "saved_at": int(time.time())}
        self._save(data)

    def clear(self):
        if self._path.exists():
            self._path.unlink()
```

```python
# opscli/core/system_registry.py
import json
from pathlib import Path
from opscli.exceptions import SystemNotFoundError

class SystemRegistry:
    def __init__(self, base_dir: Path | None = None, builtin_systems: list | None = None):
        base = Path(base_dir or (Path.home() / ".config" / "opscli"))
        base.mkdir(parents=True, exist_ok=True)
        self._path = base / "systems.json"
        self._builtin = {s["alias"]: s for s in (builtin_systems or [])}

    def _load(self) -> list:
        return json.loads(self._path.read_text()) if self._path.exists() else []

    def _save(self, systems: list):
        self._path.write_text(json.dumps(systems, ensure_ascii=False, indent=2))

    def list_all(self) -> list:
        stored = {s["alias"]: s for s in self._load()}
        return list({**self._builtin, **stored}.values())

    def get(self, alias: str) -> dict:
        for s in self.list_all():
            if s["alias"] == alias:
                return s
        raise SystemNotFoundError(f"系统 '{alias}' 未注册")

    def add_local(self, alias: str, system_key: str, url: str,
                  token_endpoint: str = "/api/auth/cli-token"):
        systems = [s for s in self._load() if s["alias"] != alias]
        systems.append({"alias": alias, "system_key": system_key, "url": url,
                        "token_endpoint": token_endpoint, "source": "local"})
        self._save(systems)

    def remove(self, alias: str):
        if alias in self._builtin:
            raise SystemNotFoundError(f"内置系统 '{alias}' 不可删除")
        self._save([s for s in self._load() if s["alias"] != alias])

    def sync_from_ops(self, ops_systems: list):
        existing = {s["alias"]: s for s in self._load()}
        kept = {a: s for a, s in existing.items() if s.get("source") != "ops_sync"}
        for s in ops_systems:
            if s["alias"] not in kept:
                kept[s["alias"]] = {**s, "source": "ops_sync"}
        self._save(list(kept.values()))
```

```python
# opscli/core/token_manager.py
import httpx
from datetime import datetime, timezone, timedelta
from opscli.exceptions import NotAuthenticatedError, TokenFetchError

REFRESH_THRESHOLD = 300  # 距过期 5 分钟内触发刷新

class TokenManager:
    def __init__(self, store, registry):
        self._store = store
        self._registry = registry

    def _get_session_id(self) -> str:
        data = self._store.load()
        if not data or not data.get("session_id"):
            raise NotAuthenticatedError("未登录，请运行:opscli login")
        exp = datetime.fromisoformat(data["session_expires_at"])
        if exp < datetime.now(timezone.utc):
            raise NotAuthenticatedError("登录已过期，请重新运行:opscli login")
        return data["session_id"]

    def _is_valid(self, token_data: dict) -> bool:
        try:
            exp = datetime.fromisoformat(token_data["expires_at"])
            return exp > datetime.now(timezone.utc) + timedelta(seconds=REFRESH_THRESHOLD)
        except (KeyError, ValueError):
            return False

    def _fetch_token(self, system_key: str, url: str, endpoint: str) -> str:
        session_id = self._get_session_id()
        try:
            resp = httpx.post(f"{url}{endpoint}",
                              json={"session_id": session_id}, timeout=10)
            resp.raise_for_status()
            body = resp.json()
            self._store.save_token(system_key, body["jwt"],
                                   body.get("expires_in", 7200))
            return body["jwt"]
        except httpx.HTTPStatusError as e:
            raise TokenFetchError(f"获取 {system_key} JWT 失败: {e.response.status_code}")
        except Exception as e:
            raise TokenFetchError(f"获取 {system_key} JWT 异常: {e}")

    def get_token(self, alias: str) -> str:
        sys = self._registry.get(alias)
        data = self._store.load() or {}
        td = data.get("tokens", {}).get(sys["system_key"])
        if td and self._is_valid(td):
            return td["jwt"]
        return self._fetch_token(sys["system_key"], sys["url"], sys["token_endpoint"])

    def check_token(self, alias: str) -> dict:
        sys = self._registry.get(alias)
        data = self._store.load() or {}
        td = data.get("tokens", {}).get(sys["system_key"])
        if not td:
            return {"valid": False, "expires_in": 0}
        try:
            exp = datetime.fromisoformat(td["expires_at"])
            rem = int((exp - datetime.now(timezone.utc)).total_seconds())
            return {"valid": rem > 0, "expires_in": max(0, rem)}
        except (KeyError, ValueError):
            return {"valid": False, "expires_in": 0}

    def refresh_token(self, alias: str) -> str:
        sys = self._registry.get(alias)
        return self._fetch_token(sys["system_key"], sys["url"], sys["token_endpoint"])

    def refresh_all(self) -> dict:
        results = {}
        for sys in self._registry.list_all():
            try:
                self._fetch_token(sys["system_key"], sys["url"], sys["token_endpoint"])
                results[sys["alias"]] = "ok"
            except Exception as e:
                results[sys["alias"]] = f"失败: {e}"
        return results
```

```python
import time, httpx
from opscli.exceptions import DeviceFlowExpiredError, DeviceFlowDeniedError

class DeviceFlow:
    def __init__(self, ops_url: str, store):
        self._url = ops_url.rstrip("/")
        self._store = store

    def request_device_code(self) -> dict:
        resp = httpx.post(f"{self._url}/v1/cli/device/code", timeout=10)
        resp.raise_for_status()
        return resp.json()

    def poll(self, device_code: str, interval: int = 3, max_wait: int = 300) -> dict:
        elapsed = 0
        while elapsed < max_wait:
            time.sleep(interval)
            elapsed += interval
            resp = httpx.get(f"{self._url}/v1/cli/device/poll",
                             params={"device_code": device_code}, timeout=10)
            resp.raise_for_status()
            body = resp.json()
            status = body.get("status")
            if status == "authorized":
                self._store.save_session(body["session_id"],
                                         body.get("email", ""),
                                         body.get("expires_at", ""))
                return body
            elif status == "expired":
                raise DeviceFlowExpiredError("设备码已超时，请重新运行:opscli login")
            elif status == "denied":
                raise DeviceFlowDeniedError("用户拒绝授权")
        raise DeviceFlowExpiredError("等待超时，请重新运行:opscli login")
```

```python
    NotAuthenticatedError, SystemNotFoundError,
    DeviceFlowExpiredError, DeviceFlowDeniedError,
)

app = typer.Typer(help="Aukeys 多系统 CLI 认证工具")
token_app = typer.Typer(help="JWT Token 管理")
system_app = typer.Typer(help="系统注册管理")
app.add_typer(token_app, name="token")
app.add_typer(system_app, name="system")

console = Console()

def _client() -> AuthClient:
    return AuthClient()

def _registry() -> SystemRegistry:
    return SystemRegistry(builtin_systems=BUILTIN_SYSTEMS)

# ── 授权管理 ──────────────────────────────────────────

@app.command()
def login():
    """发起 Device Flow 授权（自动打开浏览器）"""
    store = CredentialStore()
    flow = DeviceFlow(ops_url=OPS_URL, store=store)
    try:
        code = flow.request_device_code()
        console.print(f"\n[bold]请在浏览器打开：[/bold] {code['verification_url']}")
        console.print(f"[bold]输入验证码：  [/bold] [green]{code['user_code']}[/green]")
        console.print(f"等待授权中...（{code['expires_in']} 秒内完成）\n")
        webbrowser.open(f"{code['verification_url']}?code={code['user_code']}")
        result = flow.poll(code["device_code"], interval=code.get("interval", 3))
        # 登录后同步系统列表
        try:
            resp = httpx.get(f"{OPS_URL}/v1/cli/systems",
                             headers={"X-Session-Id": result["session_id"]}, timeout=10)
            if resp.status_code == 200:
                _registry().sync_from_ops(resp.json().get("systems", []))
        except Exception:
            pass
        console.print(f"[green]✓ 授权成功！账号：{result.get('email', '')}[/green]")
    except (DeviceFlowExpiredError, DeviceFlowDeniedError) as e:
        console.print(f"[red]✗ {e}[/red]")
        raise typer.Exit(1)

@app.command()
def logout():
    """清除本地所有凭证"""
    CredentialStore().clear()
    console.print("[green]✓ 已退出，本地凭证已清除[/green]")

@app.command()
def status():
    """查看当前登录状态与各系统 Token 情况"""
    c = _client()
    data = c._store.load()
    if not data:
        console.print("[yellow]未登录[/yellow]")
        return
    console.print(f"[green]已登录[/green]  {data.get('email', '')}")
    console.print(f"Session 过期：{data.get('session_expires_at', 'N/A')}\n")
    t = Table("别名", "系统", "Token 状态", "剩余时间(s)")
    for s in c._registry.list_all():
        r = c.check_token(s["alias"])
        t.add_row(s["alias"], s["system_key"],
                  "[green]有效[/green]" if r["valid"] else "[red]无效/未获取[/red]",
                  str(r["expires_in"]))
    console.print(t)

# ── Token 操作 ─────────────────────────────────────────

@token_app.command("get")
def token_get(system: str = typer.Option(..., "--system", "-s", help="系统别名")):
    """获取指定系统 JWT（纯文本输出，适合脚本）"""
    try:
        typer.echo(_client().get_token(system))
    except (NotAuthenticatedError, SystemNotFoundError) as e:
        console.print(f"[red]{e}[/red]", err=True)
        raise typer.Exit(1)

@token_app.command("check")
def token_check(system: str = typer.Option(..., "--system", "-s")):
    """检测指定系统 JWT 有效性"""
    r = _client().check_token(system)
    if r["valid"]:
        console.print(f"[green]✓ 有效[/green]  剩余 {r['expires_in']} 秒")
    else:
        console.print("[red]✗ 已过期或未获取[/red]")
        raise typer.Exit(1)

@token_app.command("refresh")
def token_refresh(
    system: str = typer.Option(None, "--system", "-s"),
    all_systems: bool = typer.Option(False, "--all"),
):
    """刷新 JWT（--system 指定单个，--all 刷新全部）"""
    c = _client()
    if all_systems:
        for alias, st in c._tm.refresh_all().items():
            icon = "[green]✓[/green]" if st == "ok" else "[red]✗[/red]"
            console.print(f"{icon} {alias}: {st}")
    elif system:
        try:
            c.refresh_token(system)
            console.print(f"[green]✓ {system} JWT 已刷新[/green]")
        except Exception as e:
            console.print(f"[red]✗ {e}[/red]")
            raise typer.Exit(1)
    else:
        console.print("[red]请指定 --system 或 --all[/red]")
        raise typer.Exit(1)

# ── 系统管理 ───────────────────────────────────────────

@system_app.command("list")
def system_list():
    """列出所有已注册系统"""
    t = Table("别名", "System Key", "URL", "来源")
    for s in _registry().list_all():
        t.add_row(s["alias"], s["system_key"], s["url"], s.get("source", ""))
    console.print(t)

@system_app.command("sync")
def system_sync():
    """从 ops 同步多实例系统列表"""
    data = CredentialStore().load()
    if not data:
        console.print("[red]未登录，请先运行:opscli login[/red]")
        raise typer.Exit(1)
    try:
        resp = httpx.get(f"{OPS_URL}/v1/cli/systems",
                         headers={"X-Session-Id": data["session_id"]}, timeout=10)
        resp.raise_for_status()
        systems = resp.json().get("systems", [])
        _registry().sync_from_ops(systems)
        console.print(f"[green]✓ 同步完成，共 {len(systems)} 个系统[/green]")
    except Exception as e:
        console.print(f"[red]✗ 同步失败: {e}[/red]")
        raise typer.Exit(1)

@system_app.command("add")
def system_add(
    alias: str = typer.Option(..., "--alias"),
    url: str = typer.Option(..., "--url"),
    key: str = typer.Option(None, "--key", help="存储键，默认由 alias 生成"),
):
    """手动添加系统实例（source=local）"""
    system_key = key or alias.replace(" ", "_").lower()
    _registry().add_local(alias, system_key, url)
    console.print(f"[green]✓ 已添加：{alias}[/green]")

@system_app.command("remove")
def system_remove(alias: str = typer.Option(..., "--alias")):
    """移除手动添加的系统"""
    try:
        _registry().remove(alias)
        console.print(f"[green]✓ 已移除：{alias}[/green]")
    except SystemNotFoundError as e:
        console.print(f"[red]{e}[/red]")
        raise typer.Exit(1)

@app.command()
def doctor():
    """检查登录状态与各系统连通性"""
    c = _client()
    console.print("[bold]opscli 环境检查\n[/bold]")
    if c.is_authenticated():
        console.print("[green]✓ 已登录[/green]")
    else:
        console.print("[red]✗ 未登录（运行opscli login）[/red]")
    for s in c._registry.list_all():
        try:
            httpx.get(s["url"], timeout=5)
            console.print(f"[green]✓ {s['alias']} 可访问[/green]")
        except Exception:
            console.print(f"[red]✗ {s['alias']} 不可达 ({s['url']})[/red]")
```
