```php
<?php

namespace App\Services\DataMetrics;

use Illuminate\Support\Facades\DB;

/**
 * 意图效果统计服务（只读聚合）
 *
 * 数据源：dm_intent_match_records（命中事件）+ dm_dataset_query_records（执行归因）。
 * 转化关联走 intent_match_record_id 强关联（opscli 透传），不做时间窗猜测。
 * 不建预聚合表：当前事件量级 SQL 直算足够，索引 idx_imr_intent_date / idx_dqr_intent_date 兜底。
 */
class IntentStatsService
{
    /** 按意图汇总时间窗内的命中数、执行转化数、成功数与最近命中时间。 */
    public function summary(int $days = 30): array
    {
        $since = now()->subDays($days)->toDateString();

        // 命中侧：matched=1 按意图计数
        $hits = DB::connection('ops_metrics')->table('dm_intent_match_records')
            ->selectRaw('intent_code, COUNT(*) AS hits, MAX(created_at) AS last_hit_at')
            ->where('matched', 1)
            ->where('called_date', '>=', $since)
            ->groupBy('intent_code')
            ->get()->keyBy('intent_code');

        // 执行侧：强关联记录按意图计数，同时算成功数
        $executed = DB::connection('ops_metrics')->table('dm_dataset_query_records')
            ->selectRaw('intent_code, COUNT(*) AS executed, SUM(is_success) AS success')
            ->whereNotNull('intent_match_record_id')
            ->where('called_date', '>=', $since)
            ->groupBy('intent_code')
            ->get()->keyBy('intent_code');

        $summary = [];
        foreach ($hits as $code => $hit) {
            $exec = $executed->get($code);
            $executedCount = (int) ($exec->executed ?? 0);
            $successCount = (int) ($exec->success ?? 0);
            $summary[$code] = [
                'hits' => (int) $hit->hits,
                'executed' => $executedCount,
                'success' => $successCount,
                'success_rate' => $executedCount > 0 ? round($successCount * 100.0 / $executedCount, 1) : 0.0,
                'last_hit_at' => (string) $hit->last_hit_at,
            ];
        }
        return $summary;
    }

    /** 单意图按天趋势：命中数与执行数并列。 */
    public function intentDaily(string $intentCode, int $days = 30): array
    {
        $since = now()->subDays($days)->toDateString();

        $hits = DB::connection('ops_metrics')->table('dm_intent_match_records')
            ->selectRaw('called_date, COUNT(*) AS hits')
            ->where('intent_code', $intentCode)->where('matched', 1)
            ->where('called_date', '>=', $since)
            ->groupBy('called_date')->pluck('hits', 'called_date');

        $executed = DB::connection('ops_metrics')->table('dm_dataset_query_records')
            ->selectRaw('called_date, COUNT(*) AS executed')
            ->where('intent_code', $intentCode)->whereNotNull('intent_match_record_id')
            ->where('called_date', '>=', $since)
            ->groupBy('called_date')->pluck('executed', 'called_date');

        $dates = collect($hits->keys())->merge($executed->keys())->unique()->sort()->values();
        return $dates->map(fn ($date) => [
            'date' => (string) $date,
            'hits' => (int) ($hits[$date] ?? 0),
            'executed' => (int) ($executed[$date] ?? 0),
        ])->all();
    }

    /** 未命中查询榜：按原文分组计频，指引该新增什么意图。 */
    public function unmatchedQueries(int $days = 30, int $limit = 50): array
    {
        $since = now()->subDays($days)->toDateString();
        return DB::connection('ops_metrics')->table('dm_intent_match_records')
            ->selectRaw('query_text, COUNT(*) AS cnt, MAX(created_at) AS last_seen')
            ->where('matched', 0)
            ->where('query_text', '!=', '')
            ->where('called_date', '>=', $since)
            ->groupBy('query_text')
            ->orderByDesc('cnt')->orderByDesc('last_seen')
            ->limit($limit)
            ->get()
            ->map(fn ($row) => [
                'query_text' => (string) $row->query_text,
                'cnt' => (int) $row->cnt,
                'last_seen' => (string) $row->last_seen,
            ])->all();
    }
}
```

```php
    /** 意图效果汇总（列表页聚合列数据源） */
    public function statsSummary(Request $request)
    {
        try {
            $days = min(max((int) $request->input('days', 30), 1), 365);
            return ApiResponse::success(app(\App\Services\DataMetrics\IntentStatsService::class)->summary($days));
        } catch (\Exception $e) {
            return ApiResponse::error('获取意图统计失败: ' . $e->getMessage());
        }
    }

    /** 单意图统计详情：汇总 + 按天趋势 */
    public function stats(Request $request, $id)
    {
        try {
            $intent = DB::connection('ops_metrics')->table('dm_dataset_query_intents')
                ->whereNull('deleted_at')->find($id);
            if (!$intent) {
                return ApiResponse::error('查询意图不存在', 404);
            }
            $days = min(max((int) $request->input('days', 30), 1), 365);
            $service = app(\App\Services\DataMetrics\IntentStatsService::class);
            return ApiResponse::success([
                'summary' => $service->summary($days)[$intent->intent_code] ?? [
                    'hits' => 0, 'executed' => 0, 'success' => 0, 'success_rate' => 0.0, 'last_hit_at' => '',
                ],
                'daily' => $service->intentDaily($intent->intent_code, $days),
            ]);
        } catch (\Exception $e) {
            return ApiResponse::error('获取意图统计详情失败: ' . $e->getMessage());
        }
    }

    /** 未命中查询榜：指引该新增什么意图 */
    public function unmatchedQueries(Request $request)
    {
        try {
            $days = min(max((int) $request->input('days', 30), 1), 365);
            $limit = min(max((int) $request->input('limit', 50), 1), 200);
            return ApiResponse::success(
                app(\App\Services\DataMetrics\IntentStatsService::class)->unmatchedQueries($days, $limit)
            );
        } catch (\Exception $e) {
            return ApiResponse::error('获取未命中查询榜失败: ' . $e->getMessage());
        }
    }
```

```javascript
        async function loadUnmatched() {
            const tbody = document.getElementById('unmatchedTable');
            tbody.innerHTML = '<tr><td colspan="4" class="text-center text-muted py-3">加载中...</td></tr>';
            try {
                const resp = await fetch(`${API_BASE}/unmatched-queries?days=30&limit=50`, { headers: apiHeaders() });
                const result = await resp.json();
                const items = (result.code === 200 ? result.data : []) || [];
                tbody.innerHTML = items.length
                    ? items.map((row, i) => `
                        <tr>
                            <td>${i + 1}</td>
                            <td>${escapeHtml(row.query_text)}</td>
                            <td><span class="badge bg-secondary">${row.cnt}</span></td>
                            <td class="text-muted small">${row.last_seen}</td>
                        </tr>`).join('')
                    : '<tr><td colspan="4" class="text-center text-muted py-3">近30天没有未命中查询</td></tr>';
            } catch (e) {
                tbody.innerHTML = `<tr><td colspan="4" class="text-center text-danger py-3">加载失败: ${e.message}</td></tr>`;
            }
        }

        // 未命中榜展示用户原文，必须转义防注入
        function escapeHtml(text) {
            const div = document.createElement('div');
            div.textContent = text ?? '';
            return div.innerHTML;
        }
```
