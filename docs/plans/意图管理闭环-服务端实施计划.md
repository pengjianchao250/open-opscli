`src/Services/IntentMatchRecordService.php`：

```php
<?php

namespace Aukey\DataMetrics\Services;

use Illuminate\Support\Facades\DB;

/**
 * 意图匹配事件落库服务
 *
 * 只做一件事：把客户端上报的匹配事件写入 dm_intent_match_records 并返回记录 id。
 * request_id 的取值逻辑与主仓库 DatasetQueryRecordService 保持一致（LogRequest UUID 优先），
 * 保证同一请求链上 match 记录与查询记录可经 request_id 交叉核对。
 */
class IntentMatchRecordService
{
    public function record(array $payload, int $userId): int
    {
        return (int) DB::connection('ops_metrics')->table('dm_intent_match_records')->insertGetId([
            'request_id' => mb_substr($this->resolveRequestId(), 0, 64),
            'intent_code' => mb_substr((string) ($payload['intent_code'] ?? ''), 0, 80),
            'matched' => (bool) ($payload['matched'] ?? false),
            'fallback_reason' => mb_substr((string) ($payload['fallback_reason'] ?? ''), 0, 32),
            'match_source' => mb_substr((string) ($payload['match_source'] ?? ''), 0, 32),
            'score' => (int) ($payload['score'] ?? 0),
            'ask_required' => (bool) ($payload['ask_required'] ?? false),
            'query_text' => mb_substr((string) ($payload['query_text'] ?? ''), 0, 500),
            'query_keywords' => json_encode(
                is_array($payload['query_keywords'] ?? null) ? $payload['query_keywords'] : [],
                JSON_UNESCAPED_UNICODE
            ),
            'catalog_version' => mb_substr((string) ($payload['catalog_version'] ?? ''), 0, 32),
            'user_id' => $userId,
            'called_date' => now()->toDateString(),
            'created_at' => now(),
        ]);
    }

    /** 与主仓库 DatasetQueryRecordService::resolveRequestId() 同口径取请求标识 */
    private function resolveRequestId(): string
    {
        $requestId = request()->attributes->get('log_request_uuid', '');
        if (empty($requestId)) {
            $requestId = (string) request()->header('X-Request-ID', '');
        }
        return $requestId;
    }
}
```

- [ ] **Step 4: 实现 Controller 方法与路由**

`DatasetSkillApiController` 在 `catalog()` 后追加（构造器注入或方法内 `app()` 解析，跟随该控制器现有依赖注入风格）：

```php
    /**
     * 接收客户端意图匹配事件上报（含未命中事件）。
     *
     * POST /api/v1/data-metrics/datasets/skill/intent-match-report
     */
    public function reportIntentMatch(Request $request): JsonResponse
    {
        $validator = Validator::make($request->all(), [
            'matched' => ['required', 'boolean'],
            'intent_code' => ['nullable', 'string', 'max:80'],
            'score' => ['nullable', 'integer'],
            'ask_required' => ['nullable', 'boolean'],
            'fallback_reason' => ['nullable', 'string', 'max:32'],
            'match_source' => ['required', 'string', 'max:32'],
            'query_text' => ['nullable', 'string'],
            'query_keywords' => ['nullable', 'array'],
            'catalog_version' => ['nullable', 'string', 'max:32'],
        ]);
        if ($validator->fails()) {
            return ApiResponse::validationError($validator->errors()->toArray(), '参数验证失败');
        }

        try {
            $userId = auth()->id();
            if (!$userId) {
                return ApiResponse::unauthorized('未授权访问');
            }
            $recordId = app(\Aukey\DataMetrics\Services\IntentMatchRecordService::class)
                ->record($validator->validated(), (int) $userId);

            return ApiResponse::success(['match_record_id' => $recordId]);
        } catch (\Throwable $e) {
            return ApiResponse::serverError('意图匹配事件上报失败: ' . $e->getMessage());
        }
    }
```
